#### TC-SSE-003: TTFT (Time to First Token) 测量

**目的**: 测量首个 token 到达时间

**指标**:
| 指标 | 目标值 |
|------|--------|
| TTFT P50 | < 500ms |
| TTFT P95 | < 1000ms |
| TTFT P99 | < 2000ms |

**测试代码**:
```go
func TestTTFT(t *testing.T) {
    var ttftSamples []time.Duration

    for i := 0; i < 100; i++ {
        start := time.Now()
        resp, _ := sendStreamRequest(createTestRequest())

        scanner := bufio.NewScanner(resp.Body)
        for scanner.Scan() {
            if strings.HasPrefix(scanner.Text(), "data: ") {
                ttft := time.Since(start)
                ttftSamples = append(ttftSamples, ttft)
                break
            }
        }
        resp.Body.Close()
    }

    // 计算百分位数
    sort.Slice(ttftSamples, func(i, j int) bool {
        return ttftSamples[i] < ttftSamples[j]
    })

    p50 := ttftSamples[50]
    p95 := ttftSamples[95]
    p99 := ttftSamples[99]

    t.Logf("TTFT P50: %v, P95: %v, P99: %v", p50, p95, p99)

    assert.Less(t, p50, 500*time.Millisecond)
    assert.Less(t, p95, 1000*time.Millisecond)
}
```

**测试脚本**:
```bash
#!/bin/bash
# daemon_stability_test.sh

DURATION=86400  # 24 hours
INTERVAL=60     # 1 minute

start_time=$(date +%s)
end_time=$((start_time + DURATION))

while [ $(date +%s) -lt $end_time ]; do
    # 健康检查
    status=$(curl -s -o /dev/null -w "%{http_code}" http://127.0.0.1:${PORT}/api/v1/proxy/health)

    # 内存使用
    mem=$(ps -o rss= -p $PID)

    # CPU 使用
    cpu=$(ps -o %cpu= -p $PID)

    echo "$(date): status=$status, mem=${mem}KB, cpu=${cpu}%"

    sleep $INTERVAL
done
```

```go
func TestRetryMechanism(t *testing.T) {
    tests := []struct {
        errorCode    string
        shouldRetry  bool
        maxRetries   int
    }{
        {Error500, true, 3},
        {Error502, true, 3},
        {Error429, true, 3},
        {"400", false, 0},
    }

    for _, tt := range tests {
        t.Run(tt.errorCode, func(t *testing.T) {
            mockServer.SetErrorMode(tt.errorCode)
            mockServer.ResetRequestCount()

            sendRequest(createTestRequest())

            expectedCalls := 1
            if tt.shouldRetry {
                expectedCalls = tt.maxRetries + 1
            }
            assert.Equal(t, expectedCalls, mockServer.GetRequestCount())
        })
    }
}
```

```go
    for _, tc := range testCases {
```

## Mock Server 实现

### 完整 Mock Server 代码

```go
// internal/proxy/testutil/mock_server.go

package testutil

import (
    "encoding/json"
    "fmt"
    "net/http"
    "sync"
    "sync/atomic"
    "time"
)

type MockServer struct {
    server           *http.Server
    port             int
    mu               sync.RWMutex

    // 配置
    responseDelay    time.Duration
    streamChunkDelay time.Duration
    errorMode        string
    statusCode       int
    healthy          bool

    // 统计
    requestCount     int64

    // SSE 控制
    interruptAfter   int
    malformedChunkAt int
    tokenUsage       TokenUsage
}

type TokenUsage struct {
    PromptTokens     int `json:"prompt_tokens"`
    CompletionTokens int `json:"completion_tokens"`
    TotalTokens      int `json:"total_tokens"`
}

func NewMockServer(port int) *MockServer {
    ms := &MockServer{
        port:       port,
        statusCode: 200,
        healthy:    true,
    }

    mux := http.NewServeMux()
    mux.HandleFunc("/v1/chat/completions", ms.handleChatCompletions)
    mux.HandleFunc("/v1/models", ms.handleModels)
    mux.HandleFunc("/health", ms.handleHealth)

    ms.server = &http.Server{
        Addr:    fmt.Sprintf(":%d", port),
        Handler: mux,
    }

    return ms
}

func (ms *MockServer) Start() error {
    go ms.server.ListenAndServe()
    return nil
}

func (ms *MockServer) Stop() error {
    return ms.server.Close()
}

func (ms *MockServer) handleChatCompletions(w http.ResponseWriter, r *http.Request) {
    atomic.AddInt64(&ms.requestCount, 1)

    ms.mu.RLock()
    delay := ms.responseDelay
    errorMode := ms.errorMode
    statusCode := ms.statusCode
    streamDelay := ms.streamChunkDelay
    interruptAfter := ms.interruptAfter
    malformedAt := ms.malformedChunkAt
    tokenUsage := ms.tokenUsage
    ms.mu.RUnlock()

    // 应用延迟
    if delay > 0 {
        time.Sleep(delay)
    }

    // 处理错误模式
    if errorMode != "" && errorMode != "none" {
        ms.handleError(w, errorMode)
        return
    }

    if statusCode != 200 {
        w.WriteHeader(statusCode)
        return
    }

    // 检查是否是流式请求
    var req map[string]interface{}
    json.NewDecoder(r.Body).Decode(&req)

    if stream, ok := req["stream"].(bool); ok && stream {
        ms.handleStreamResponse(w, streamDelay, interruptAfter, malformedAt, tokenUsage)
    } else {
        ms.handleNonStreamResponse(w, tokenUsage)
    }
}

func (ms *MockServer) handleStreamResponse(w http.ResponseWriter, chunkDelay time.Duration,
    interruptAfter, malformedAt int, usage TokenUsage) {
    w.Header().Set("Content-Type", "text/event-stream")
    w.Header().Set("Cache-Control", "no-cache")
    w.Header().Set("Connection", "keep-alive")
    w.WriteHeader(200)

    flusher, ok := w.(http.Flusher)
    if !ok {
        return
    }

    // 模拟逐字输出
    content := "This is a test response from the mock server."
    chunks := splitIntoChunks(content, 5) // 每 5 个字符一个 chunk

    for i, chunk := range chunks {
        if interruptAfter > 0 && i >= interruptAfter {
            return // 模拟中断
        }

        if malformedAt > 0 && i == malformedAt {
            fmt.Fprintf(w, "data: {invalid json}\n\n")
            flusher.Flush()
            continue
        }

        data := map[string]interface{}{
            "id":      fmt.Sprintf("chunk-%d", i),
            "object":  "chat.completion.chunk",
            "created": time.Now().Unix(),
            "choices": []map[string]interface{}{
                {
                    "index": 0,
                    "delta": map[string]string{
                        "content": chunk,
                    },
                },
            },
        }

        jsonData, _ := json.Marshal(data)
        fmt.Fprintf(w, "data: %s\n\n", jsonData)
        flusher.Flush()

        if chunkDelay > 0 {
            time.Sleep(chunkDelay)
        }
    }

    // 发送最终 chunk 带 usage
    finalData := map[string]interface{}{
        "id":      "final",
        "object":  "chat.completion.chunk",
        "created": time.Now().Unix(),
        "choices": []map[string]interface{}{
            {
                "index":         0,
                "delta":         map[string]string{},
                "finish_reason": "stop",
            },
        },
        "usage": usage,
    }
    jsonData, _ := json.Marshal(finalData)
    fmt.Fprintf(w, "data: %s\n\n", jsonData)
    fmt.Fprintf(w, "data: [DONE]\n\n")
    flusher.Flush()
}

func (ms *MockServer) handleNonStreamResponse(w http.ResponseWriter, usage TokenUsage) {
    response := map[string]interface{}{
        "id":      "resp-123",
        "object":  "chat.completion",
        "created": time.Now().Unix(),
        "choices": []map[string]interface{}{
            {
                "index": 0,
                "message": map[string]string{
                    "role":    "assistant",
                    "content": "This is a test response.",
                },
                "finish_reason": "stop",
            },
        },
        "usage": usage,
    }

    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(response)
}

func (ms *MockServer) handleError(w http.ResponseWriter, errorMode string) {
    switch errorMode {
    case "500":
        w.WriteHeader(500)
        json.NewEncoder(w).Encode(map[string]string{"error": "Internal Server Error"})
    case "502":
        w.WriteHeader(502)
        json.NewEncoder(w).Encode(map[string]string{"error": "Bad Gateway"})
    case "503":
        w.WriteHeader(503)
        json.NewEncoder(w).Encode(map[string]string{"error": "Service Unavailable"})
    case "429":
        w.Header().Set("Retry-After", "60")
        w.WriteHeader(429)
        json.NewEncoder(w).Encode(map[string]string{"error": "Rate Limited"})
    case "timeout":
        time.Sleep(120 * time.Second) // 超过客户端超时
    case "conn_reset":
        // 强制关闭连接 (需要底层支持)
        if hijacker, ok := w.(http.Hijacker); ok {
            conn, _, _ := hijacker.Hijack()
            conn.Close()
        }
    }
}

func (ms *MockServer) handleModels(w http.ResponseWriter, r *http.Request) {
    response := map[string]interface{}{
        "object": "list",
        "data": []map[string]interface{}{
            {"id": "mock-model", "object": "model"},
        },
    }
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(response)
}

func (ms *MockServer) handleHealth(w http.ResponseWriter, r *http.Request) {
    ms.mu.RLock()
    healthy := ms.healthy
    ms.mu.RUnlock()

    if healthy {
        w.WriteHeader(200)
        json.NewEncoder(w).Encode(map[string]string{"status": "healthy"})
    } else {
        w.WriteHeader(503)
        json.NewEncoder(w).Encode(map[string]string{"status": "unhealthy"})
    }
}

// 配置方法
func (ms *MockServer) SetErrorMode(mode string)           { ms.mu.Lock(); ms.errorMode = mode; ms.mu.Unlock() }
func (ms *MockServer) SetStatusCode(code int)             { ms.mu.Lock(); ms.statusCode = code; ms.mu.Unlock() }
func (ms *MockServer) SetResponseDelay(d time.Duration)   { ms.mu.Lock(); ms.responseDelay = d; ms.mu.Unlock() }
func (ms *MockServer) SetStreamChunkDelay(d time.Duration){ ms.mu.Lock(); ms.streamChunkDelay = d; ms.mu.Unlock() }
func (ms *MockServer) SetHealthy(healthy bool)            { ms.mu.Lock(); ms.healthy = healthy; ms.mu.Unlock() }
func (ms *MockServer) SetInterruptAfterChunks(n int)      { ms.mu.Lock(); ms.interruptAfter = n; ms.mu.Unlock() }
func (ms *MockServer) SetMalformedChunkAt(n int)          { ms.mu.Lock(); ms.malformedChunkAt = n; ms.mu.Unlock() }
func (ms *MockServer) SetTokenUsage(usage TokenUsage)     { ms.mu.Lock(); ms.tokenUsage = usage; ms.mu.Unlock() }
func (ms *MockServer) GetRequestCount() int64             { return atomic.LoadInt64(&ms.requestCount) }
func (ms *MockServer) ResetRequestCount()                 { atomic.StoreInt64(&ms.requestCount, 0) }

func splitIntoChunks(s string, size int) []string {
    var chunks []string
    for i := 0; i < len(s); i += size {
        end := i + size
        if end > len(s) {
            end = len(s)
        }
        chunks = append(chunks, s[i:end])
    }
    return chunks
}
```
